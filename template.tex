\begin{definition}[Kernel]\label{definition:kernel}
For an \(R\)-linear map \(f:M\to N\), define
\[
    \ker f=\set{m\in M\mid f(m)=0}.
\]
\end{definition}

\begin{thm}[First isomorphism theorem]\label{thm:first}
The map \(M/\ker f\to\im f\), \(m+\ker f\mapsto f(m)\),
is an isomorphism of \(R\)-modules.
\end{thm}
\begin{proof}
The map is well-defined and linear, injective, and surjective.
% For a proof ending in a display, put \qedhere inside that display.
\end{proof}

\begin{lem}[Zero kernel]\label{lem:kernel}
A module homomorphism is injective if and only if its kernel is zero.
\end{lem}

\begin{prop}[Cokernel]\label{prop:coker}
The quotient \(\coker f=N/\im f\) is an \(R\)-module.
\end{prop}

\begin{cor}[Injective maps]\label{cor:injective}
If \(f\) is injective, then \(M\cong\im f\).
\end{cor}

\begin{ex}[Multiplication by two]
For \(f:\ZZ\to\ZZ\), \(n\mapsto2n\), we have
\(\ker f=0\), \(\im f=2\ZZ\), and \(\coker f\cong\ZZ/2\ZZ\).
\end{ex}

\begin{prob*}[An exercise]
Write down the isomorphism in \cref{thm:first} for the preceding example.
\end{prob*}

Use \verb|\cref{thm:first}| within a sentence and
\verb|\Cref{thm:first}| at its beginning.
For example: \Cref{thm:first} implies \cref{cor:injective}.
\end{remark*}
